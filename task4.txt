select users.*
from users
join orders on orders.user_id = users.id
where users.birthday_at <= CURDATE() - INTERVAL 30 YEAR and orders.created_at >= CURDATE() - INTERVAL 6 MONTH
group by users.id
having count(orders.id) >= 3
order by RAND()
limit 1;
